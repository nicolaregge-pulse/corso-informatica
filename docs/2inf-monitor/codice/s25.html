Il tuo salvadanaio, sistemato

Il tuo Button3 (la sfida) l'ho lasciato com'è. Ho scritto il FOR nel Button1 e il WHILE nel Button2.

  1. Premi il bottone blu Copia tutto il codice.
  2. In Lazarus clicca dentro il codice di unit1.pas, premi Ctrl+A e poi Ctrl+V.
  3. Premi F9 e prova i 3 bottoni.
  4. Salva con Ctrl+Maiusc+S e ricarica la cartella «salvadanaio» su GitHub (Add file, Upload files, Commit changes).
  5. Leggi i commenti verdi con //: devi saper spiegare a voce che cosa fa ogni riga.

unit Unit1;

{$mode objfpc}{$H+}

interface

uses
  Classes, SysUtils, Forms, Controls, Graphics, Dialogs, StdCtrls;

type

  { TForm1 }

  TForm1 = class(TForm)
    Button1: TButton;
    Button2: TButton;
    Button3: TButton;
    Edit1: TEdit;
    Edit2: TEdit;
    Label1: TLabel;
    Label2: TLabel;
    Memo1: TMemo;
    procedure Button1Click(Sender: TObject);
    procedure Button2Click(Sender: TObject);
    procedure Button3Click(Sender: TObject);
  private

  public

  end;

var
  Form1: TForm1;

implementation

{$R *.lfm}

{ TForm1 }

procedure TForm1.Button1Click(Sender: TObject);
var giorno, euro, totale: integer;
begin
  Randomize;
  Memo1.Clear;
  totale := 30;
  // ciclo FOR: giorno da 1 a 7
  for giorno := 1 to 7 do
  begin
    euro := Random(5) + 1;          // euro a caso da 1 a 5
    totale := totale + euro;        // aggiungo gli euro al totale
    Memo1.Lines.Add('Giorno ' + IntToStr(giorno) + ': +' + IntToStr(euro) + ' euro');
  end;
  // dopo il ciclo: il totale della settimana
  Memo1.Lines.Add('Totale della settimana: ' + IntToStr(totale) + ' euro');
end;

procedure TForm1.Button2Click(Sender: TObject);
var giorni, euro, totale, prezzo: integer;
begin
  Randomize;
  prezzo := StrToInt(Edit2.Text);
  Memo1.Clear;
  totale := 30;
  giorni := 0;
  // ciclo WHILE: finché il totale è minore del prezzo
  while totale < prezzo do
  begin
    euro := Random(5) + 1;          // euro a caso da 1 a 5
    totale := totale + euro;        // aggiungo gli euro al totale
    giorni := giorni + 1;           // un giorno in più
  end;
  // dopo il ciclo: quanti giorni servono
  Memo1.Lines.Add('Per ' + Edit1.Text + ' ti servono ' + IntToStr(giorni) + ' giorni');
  Memo1.Lines.Add('Cioè ' + IntToStr(giorni div 7) + ' settimane e ' + IntToStr(giorni mod 7) + ' giorni');
end;

  procedure TForm1.Button3Click(Sender: TObject);
  var
    giorni, euro, totale, prezzo: integer;
  begin
    Randomize;
    prezzo := StrToInt(Edit2.Text);
    Memo1.Clear;
    totale := 0;
    giorni := 0;

    // Ciclo WHILE
    while totale < prezzo do
    begin
      euro := Random(5) + 1;
      totale := totale + euro;
      giorni := giorni + 1;

      // Controlla se è sabato
      if giorni mod 7 = 0 then
      begin
        totale := totale + 10;
        Memo1.Lines.Add('Sabato: paghetta +10 euro!');
      end;

      Memo1.Lines.Add('Giorno ' + IntToStr(giorni) + ': ho ' + IntToStr(totale) + ' euro');
    end;

    // Risultato finale
    Memo1.Lines.Add('Per ' + Edit1.Text + ' ti servono ' + IntToStr(giorni) + ' giorni');
    Memo1.Lines.Add('Cioè ' + IntToStr(giorni div 7) + ' settimane e ' + IntToStr(giorni mod 7) + ' giorni');
  end;



end.
